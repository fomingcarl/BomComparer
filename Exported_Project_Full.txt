using System.Collections.Generic;
using System.Linq;
using System.Windows;
using BomComparer.Models;

namespace BomComparer
{
    public partial class FrequencyMismatchWindow : Window
    {
        public FrequencyMismatchWindow(string partNo, Dictionary<string, List<BomItem>> loadedBoms)
        {
            InitializeComponent();

            // Updated the title to reflect its new universal purpose
            Title = $"Parent Hierarchy Details - Part: {partNo}";

            var combinedHierarchy = new List<ParentInfo>();

            foreach (var kvp in loadedBoms)
            {
                var fileHierarchy = GetParentHierarchy(partNo, kvp.Value, kvp.Key);
                combinedHierarchy.AddRange(fileHierarchy);
            }

            int totalFiles = loadedBoms.Count;

            foreach (var item in combinedHierarchy)
            {
                int fullMatchCount = combinedHierarchy
                    .Where(x => x.ParentPartNo == item.ParentPartNo && x.ParentDescription == item.ParentDescription && x.ParentRevision == item.ParentRevision)
                    .Select(x => x.SourceFile)
                    .Distinct()
                    .Count();

                int partialMatchCount = combinedHierarchy
                    .Where(x => x.ParentPartNo == item.ParentPartNo && x.ParentDescription == item.ParentDescription)
                    .Select(x => x.SourceFile)
                    .Distinct()
                    .Count();

                if (fullMatchCount == totalFiles)
                    item.RowColor = "#C8E6C9";
                else if (partialMatchCount == totalFiles)
                    item.RowColor = "#FFF9C4";
                else
                    item.RowColor = "Transparent";
            }

            UnifiedDataGrid.ItemsSource = combinedHierarchy.OrderBy(h => h.ParentPartNo).ThenBy(h => h.SourceFile).ToList();
        }

        private List<ParentInfo> GetParentHierarchy(string partNo, List<BomItem> bom, string sourceFile)
        {
            var results = new List<ParentInfo>();
            var occurrences = bom.Where(b => b.PartNo == partNo).ToList();

            foreach (var item in occurrences)
            {
                string parentId = GetParentId(item.Id);
                var parentItem = bom.FirstOrDefault(b => b.Id == parentId);

                // Calculate the parent's level by counting the segments in its ID. If it has no parent, return "-"
                string parentLevel = string.IsNullOrEmpty(parentId) ? "-" : parentId.Split('.').Length.ToString();

                results.Add(new ParentInfo
                {
                    SourceFile = sourceFile,
                    InstanceId = item.Id,
                    ParentLevel = parentLevel, // Bind the new calculation
                    ParentPartNo = parentItem?.PartNo ?? "TOP LEVEL",
                    ParentDescription = parentItem?.Description ?? "Assembly Root",
                    ParentRevision = parentItem?.Revision ?? "-"
                });
            }
            return results;
        }

        private string GetParentId(string id)
        {
            if (string.IsNullOrEmpty(id)) return "";
            int lastDot = id.LastIndexOf('.');
            if (lastDot > 0) return id.Substring(0, lastDot);
            return "";
        }
    }

    public class ParentInfo
    {
        public string SourceFile { get; set; }
        public string InstanceId { get; set; }
        public string ParentLevel { get; set; } // Added property
        public string ParentPartNo { get; set; }
        public string ParentDescription { get; set; }
        public string ParentRevision { get; set; }
        public string RowColor { get; set; } = "Transparent";
    }
}

using System;
using System.Threading.Tasks;
using System.Windows;
using Microsoft.Win32;
using BomComparer.Services;

namespace BomComparer
{
    public partial class ReconciliationWindow : Window
    {
        private readonly IExcelService _excelService;
        private readonly IComparisonService _comparisonService;

        public ReconciliationWindow(IExcelService excelService, IComparisonService comparisonService)
        {
            InitializeComponent();
            _excelService = excelService;
            _comparisonService = comparisonService;
        }

        private void BrowseSchematic_Click(object sender, RoutedEventArgs e)
        {
            var fd = new OpenFileDialog { Filter = "Excel Files|*.xlsx" };
            if (fd.ShowDialog() == true) SchematicPathBox.Text = fd.FileName;
        }

        private void BrowseExploded_Click(object sender, RoutedEventArgs e)
        {
            var fd = new OpenFileDialog { Filter = "Excel Files|*.xlsx" };
            if (fd.ShowDialog() == true) ExplodedPathBox.Text = fd.FileName;
        }

        private async void Generate_Click(object sender, RoutedEventArgs e)
        {
            if (string.IsNullOrWhiteSpace(SchematicPathBox.Text) || string.IsNullOrWhiteSpace(ExplodedPathBox.Text))
            {
                MessageBox.Show("Please select both files before generating.", "Missing Files", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            var saveFd = new SaveFileDialog { Filter = "Excel Files|*.xlsx", DefaultExt = "xlsx", FileName = "Subsystem Cable BOM Reconciliation" };
            if (saveFd.ShowDialog() != true) return;

            GenerateButton.IsEnabled = false;
            StatusText.Text = "Processing... Please wait.";

            // 1. Extract UI strings on the main thread BEFORE starting the background task
            string schematicPath = SchematicPathBox.Text;
            string explodedPath = ExplodedPathBox.Text;
            string savePath = saveFd.FileName;

            try
            {
                await Task.Run(() =>
                {
                    // 2. Use the extracted string variables, not the UI controls
                    var schematic = _excelService.ReadSchematic(schematicPath);
                    var explodedResult = _excelService.ReadBom(explodedPath);

                    // 3. Execute Math and Categorization
                    var reportData = _comparisonService.ReconcileBoms(schematic, explodedResult.BomItems);

                    // 4. Export to Excel
                    _excelService.ExportReconciliationReport(savePath, reportData);
                });

                MessageBox.Show("Reconciliation Report generated successfully!", "Success", MessageBoxButton.OK, MessageBoxImage.Information);
                this.Close(); // Closes the dialog automatically on success
            }
            catch (Exception ex)
            {
                MessageBox.Show(ex.Message, "Error Generating Report", MessageBoxButton.OK, MessageBoxImage.Error);
                StatusText.Text = "Failed.";
            }
            finally
            {
                GenerateButton.IsEnabled = true;
            }
        }
    }
}

// Location: Services/ComparisonService.cs
using System.Collections.Generic;
using System.Linq;
using BomComparer.Models;

namespace BomComparer.Services
{
    public class ComparisonService : IComparisonService
    {
        public List<BomDifference> CompareBoms(Dictionary<string, List<BomItem>> loadedBoms)
        {
            var differences = new List<BomDifference>();

            // Requires at least 2 files to perform a comparison
            if (loadedBoms.Count < 2) return differences;

            var fileNames = loadedBoms.Keys.ToList();

            // 1. Gather every unique Part Number across all queued files
            var allPartNumbers = loadedBoms.Values.SelectMany(b => b.Select(i => i.PartNo)).Distinct().ToList();

            foreach (var partNo in allPartNumbers)
            {
                // Extract the specific BOM Items for this part from every file
                var itemsPerFile = new Dictionary<string, List<BomItem>>();
                foreach (var file in fileNames)
                {
                    itemsPerFile[file] = loadedBoms[file].Where(i => i.PartNo == partNo).ToList();
                }

                // Count how many files actually contain this part
                int filesContainingPart = itemsPerFile.Count(kvp => kvp.Value.Any());

                // Grab description and cost from the first file that has the part to populate the DataGrid
                var firstPresentItem = itemsPerFile.First(kvp => kvp.Value.Any()).Value.First();
                string desc = firstPresentItem.Description;
                double? cost = firstPresentItem.Cost;

                // PRESENCE MISMATCH: Replaces "Added/Removed" for N-Way comparisons
                if (filesContainingPart < fileNames.Count)
                {
                    var presentIn = itemsPerFile.Where(kvp => kvp.Value.Any()).Select(k => k.Key);
                    var missingFrom = itemsPerFile.Where(kvp => !kvp.Value.Any()).Select(k => k.Key);

                    differences.Add(new BomDifference
                    {
                        PartNo = partNo,
                        Description = desc,
                        Cost = cost,
                        IssueType = "Presence Mismatch",
                        Details = $"Present: [{string.Join(", ", presentIn)}]  |  Missing: [{string.Join(", ", missingFrom)}]"
                    });

                    // If a part is missing from a file entirely, comparing its quantities/costs will throw false positives. 
                    continue;
                }

                // QTY MISMATCH: Checks if any file's total Extended Qty differs from the others
                if (itemsPerFile.Values.Any(list => list.Any(i => i.ExtendedQty.HasValue)))
                {
                    var qtys = itemsPerFile.ToDictionary(k => k.Key, v => v.Value.Sum(i => i.ExtendedQty ?? 0));

                    // .Distinct() checks if there is more than 1 unique value across all files
                    if (qtys.Values.Distinct().Count() > 1)
                    {
                        string details = string.Join("  |  ", qtys.Select(q => $"[{q.Key}]: {q.Value}"));
                        differences.Add(new BomDifference { PartNo = partNo, Description = desc, Cost = cost, IssueType = "Qty Mismatch", Details = details });
                    }
                }

                // REV CHANGE
                var revs = itemsPerFile.ToDictionary(k => k.Key, v => v.Value.First().Revision);
                if (revs.Values.Distinct().Count() > 1)
                {
                    string details = string.Join("  |  ", revs.Select(r => $"[{r.Key}]: {(string.IsNullOrEmpty(r.Value) ? "-" : r.Value)}"));
                    differences.Add(new BomDifference { PartNo = partNo, Description = desc, Cost = cost, IssueType = "Rev Change", Details = details });
                }

                // UOM MISMATCH
                var uoms = itemsPerFile.ToDictionary(k => k.Key, v => v.Value.First().UoM);
                if (uoms.Values.Distinct().Count() > 1)
                {
                    string details = string.Join("  |  ", uoms.Select(u => $"[{u.Key}]: {(string.IsNullOrEmpty(u.Value) ? "-" : u.Value)}"));
                    differences.Add(new BomDifference { PartNo = partNo, Description = desc, Cost = cost, IssueType = "UoM Mismatch", Details = details });
                }

                // COST CHANGE
                if (itemsPerFile.Values.All(list => list.First().Cost.HasValue))
                {
                    var costs = itemsPerFile.ToDictionary(k => k.Key, v => v.Value.First().Cost);
                    if (costs.Values.Distinct().Count() > 1)
                    {
                        string details = string.Join("  |  ", costs.Select(c => $"[{c.Key}]: {c.Value}"));
                        differences.Add(new BomDifference { PartNo = partNo, Description = desc, Cost = cost, IssueType = "Cost Change", Details = details });
                    }
                }

                // FREQUENCY MISMATCH
                var freqs = itemsPerFile.ToDictionary(k => k.Key, v => v.Value.Count);
                if (freqs.Values.Distinct().Count() > 1)
                {
                    string details = string.Join("  |  ", freqs.Select(f => $"[{f.Key}]: {f.Value}x"));
                    differences.Add(new BomDifference { PartNo = partNo, Description = desc, Cost = cost, IssueType = "Frequency Mismatch", Details = details });
                }
            }
            return differences;
        }


        public System.Data.DataTable GenerateTopLevelReport(Dictionary<string, List<BomItem>> loadedBoms)
        {
            var dt = new System.Data.DataTable();
            if (loadedBoms.Count == 0) return dt;

            var fileNames = loadedBoms.Keys.ToList();

            // 1. Setup Static Columns
            dt.Columns.Add("Category", typeof(string));
            dt.Columns.Add("Location", typeof(string));
            dt.Columns.Add("Location by SN", typeof(string));
            dt.Columns.Add("Part Number", typeof(string));
            dt.Columns.Add("Description", typeof(string));
            dt.Columns.Add("UOM", typeof(string));
            dt.Columns.Add("Present In", typeof(string));
            dt.Columns.Add("Missing From", typeof(string));

            // 2. Setup Dynamic Columns
            foreach (var file in fileNames)
            {
                dt.Columns.Add($"{file} Qty", typeof(double));
            }
            //dt.Columns.Add("Notes", typeof(string));




            // 3. Filter all BOMs to gather unique Level 2 items (ID contains exactly 1 dot)
            var level2Parts = loadedBoms.Values
                .SelectMany(b => b.Where(i => i.Id.Count(c => c == '.') == 1))
                .Select(i => i.PartNo)
                .Distinct()
                .ToList();

            foreach (var partNo in level2Parts)
            {
                var row = dt.NewRow();
                var presentIn = new List<string>();
                var missingFrom = new List<string>();
                var locationsByFile = new Dictionary<string, string>();
                var quantities = new List<double>();

                string desc = "", uom = "";

                foreach (var file in fileNames)
                {
                    var bom = loadedBoms[file];
                    // Find the specific Level 2 item in this file
                    var item = bom.FirstOrDefault(i => i.PartNo == partNo && i.Id.Count(c => c == '.') == 1);

                    if (item != null)
                    {
                        presentIn.Add(file);
                        desc = item.Description;
                        uom = item.UoM;
                        row[$"{file} Qty"] = item.ExtendedQty ?? 0;
                        quantities.Add(item.ExtendedQty ?? 0);

                        // Find Parent Location
                        string parentId = item.Id.Split('.')[0];
                        var parent = bom.FirstOrDefault(b => b.Id == parentId);
                        string pDesc = parent?.Description?.ToUpper() ?? "";

                        if (pDesc.Contains("ASSY, CM CONFIG")) locationsByFile[file] = "CM";
                        else if (pDesc.Contains("ASSY, NEXX CONFIG")) locationsByFile[file] = "NEXX";
                        else if (pDesc.Contains("KIT, FIELD UPGRADE")) locationsByFile[file] = "FUPG";
                        else locationsByFile[file] = "UNASSIGNED";
                    }
                    else
                    {
                        missingFrom.Add(file);
                        row[$"{file} Qty"] = System.DBNull.Value;
                    }
                }

                row["Part Number"] = partNo;
                row["Description"] = desc;
                row["UOM"] = uom;
                row["Present In"] = string.Join(", ", presentIn);
                row["Missing From"] = string.Join(", ", missingFrom);

                // Location Logic
                var uniqueLocations = locationsByFile.Values.Distinct().ToList();
                if (uniqueLocations.Count == 1)
                {
                    row["Location"] = uniqueLocations.First();
                }
                else
                {
                    row["Location"] = "VARIES";
                    row["Location by SN"] = string.Join("; ", locationsByFile.Select(kvp => $"{kvp.Key}: {kvp.Value}"));
                }

                // Category Logic
                bool qtyConsistent = quantities.Distinct().Count() == 1;

                if (presentIn.Count == fileNames.Count)
                    row["Category"] = qtyConsistent ? "Common to All - Qty Consistent" : "Common to All - Qty Differs";
                else if (presentIn.Count > 1)
                    row["Category"] = qtyConsistent ? "Shared by Some - Qty Consistent" : "Shared by Some - Qty Differs";
                else
                    row["Category"] = $"Unique to {presentIn.First()}";

                dt.Rows.Add(row);
            }

            return dt;
        }



        public ReconciliationReportData ReconcileBoms(List<SchematicItem> schematic, List<BomItem> explodedBom)
        {
            var data = new ReconciliationReportData();

            // 1. Aggregate Exploded BOM to get the total sum and capture the first valid Cost and Description
            var explodedAgg = explodedBom
                .Where(b => !string.IsNullOrWhiteSpace(b.PartNo))
                .GroupBy(b => b.PartNo)
                .ToDictionary(
                    g => g.Key,
                    g => new
                    {
                        TotalQty = g.Sum(x => x.ExtendedQty ?? 0),
                        Cost = g.FirstOrDefault(x => x.Cost.HasValue)?.Cost,
                        Description = g.FirstOrDefault(x => !string.IsNullOrWhiteSpace(x.Description))?.Description
                    }
                );

            // 2. Build Main Reconciliation Sheet
            var missingItems = new List<ReconciliationItem>();

            foreach (var sch in schematic)
            {
                bool found = explodedAgg.TryGetValue(sch.PartNumber, out var agg);
                double alloc = found ? agg.TotalQty : 0;
                double remaining = sch.Qty - alloc;

                string status = "";
                if (string.IsNullOrWhiteSpace(sch.PartNumber)) status = "Manual Review";
                else if (alloc == 0) status = "Cable BOM Only";
                else if (alloc < sch.Qty) status = "Partially Accounted For";
                else status = "Fully Accounted For";

                // Contingency: If the Schematic Description is empty, try to steal it from the Exploded BOM
                string finalDesc = sch.Description;
                if (string.IsNullOrWhiteSpace(finalDesc) && found)
                    finalDesc = agg.Description ?? "";

                var rItem = new ReconciliationItem
                {
                    Subsystem = sch.Subsystem,
                    PartNumber = sch.PartNumber,
                    Description = finalDesc,
                    SchematicQty = sch.Qty,
                    AllocatedQty = alloc,
                    RemainingQty = remaining,
                    Status = status
                };
                data.MainReport.Add(rItem);

                // Filter shortages for the New Kit Adds list
                if (remaining > 0 && !string.IsNullOrWhiteSpace(sch.PartNumber))
                {
                    missingItems.Add(rItem);
                }
            }

            // 3. Build New Kit Adds Sheet (Grouped by Part)
            var groupedMissing = missingItems.GroupBy(m => m.PartNumber);
            foreach (var g in groupedMissing)
            {
                double totalAdditional = g.Sum(x => x.RemainingQty);
                var part = g.Key;

                bool found = explodedAgg.TryGetValue(part, out var agg);

                // Cost evaluates to null if missing from exploded BOM format, preventing crashes
                double? unitCost = found ? agg.Cost : null;
                double? extCost = unitCost.HasValue ? unitCost.Value * totalAdditional : null;

                string subsystems = string.Join(" | ", g.Select(x => x.Subsystem).Distinct());

                data.NewKitAdds.Add(new NewKitAdd
                {
                    PartNumber = part,
                    Description = g.First().Description,
                    AdditionalQty = totalAdditional,
                    UnitCost = unitCost,
                    ExtendedCost = extCost,
                    Subsystems = subsystems
                });
            }

            return data;
        }

        public BomImportResult ReadBom(string filePath)
        {
            var result = new BomImportResult();

            try
            {
                using (var workbook = new XLWorkbook(filePath))
                {
                    var worksheet = workbook.Worksheet(1);
                    var headerMap = new Dictionary<string, int>(StringComparer.OrdinalIgnoreCase);
                    var headerRow = worksheet.Row(1);

                    // 1. Map columns dynamically from the first row
                    foreach (var cell in headerRow.CellsUsed())
                    {
                        string headerName = cell.GetString().Trim();
                        if (!string.IsNullOrWhiteSpace(headerName))
                            headerMap[headerName] = cell.Address.ColumnNumber;
                    }

                    // --- ALIAS HELPER METHODS ---
                    // Checks multiple possible header names (aliases) and returns the first matching column index
                    int GetColumnIndex(params string[] possibleNames)
                    {
                        foreach (var name in possibleNames)
                        {
                            if (headerMap.TryGetValue(name, out int colIndex)) return colIndex;
                        }
                        return -1; // Not found
                    }

                    string GetStringSafe(IXLRow r, params string[] colNames)
                    {
                        int colIndex = GetColumnIndex(colNames);
                        return colIndex != -1 ? r.Cell(colIndex).GetString().Trim() : string.Empty;
                    }

                    double? GetDoubleSafe(IXLRow r, params string[] colNames)
                    {
                        int colIndex = GetColumnIndex(colNames);
                        if (colIndex != -1 && r.Cell(colIndex).TryGetValue<double>(out var val)) return val;
                        return null;
                    }

                    int? GetIntSafe(IXLRow r, params string[] colNames)
                    {
                        int colIndex = GetColumnIndex(colNames);
                        if (colIndex != -1 && r.Cell(colIndex).TryGetValue<int>(out var val)) return val;
                        return null;
                    }

                    // 2. Detect missing columns using the primary name and its aliases
                    void CheckRequired(string displayWarningName, params string[] aliases)
                    {
                        if (GetColumnIndex(aliases) == -1)
                            result.MissingColumns.Add(displayWarningName);
                    }

                    // Checks for either the Engineering Toolbox format or BOM Reader format
                    CheckRequired("ID or Level", "ID", "Level", "level");
                    CheckRequired("Part No", "Part No", "NPart");
                    CheckRequired("Rev", "Rev");
                    CheckRequired("Qty", "Qty", "BOM_Qty");
                    CheckRequired("UoM", "UoM", "Meas");
                    CheckRequired("Description", "Description", "Desc");
                    CheckRequired("Cost", "Cost"); // BOM Reader lacks this, so it will correctly warn and skip
                    CheckRequired("Extended Qty", "Extended Qty", "Total_Qty");

                    // 3. Read data and generate missing IDs
                    var rows = worksheet.RowsUsed().Skip(1);
                    var levelCounters = new Dictionary<int, int>(); // Tracks the numbering for the dynamic ID generator

                    foreach (var row in rows)
                    {
                        var item = new BomItem
                        {
                            PartNo = GetStringSafe(row, "Part No", "NPart"),
                            Revision = GetStringSafe(row, "Rev"),
                            Quantity = GetDoubleSafe(row, "Qty", "BOM_Qty"),
                            UoM = GetStringSafe(row, "UoM", "Meas"),
                            Description = GetStringSafe(row, "Description", "Desc"),
                            Cost = GetDoubleSafe(row, "Cost"),
                            ExtendedQty = GetDoubleSafe(row, "Extended Qty", "Total_Qty")
                        };

                        // ID GENERATION CONTINGENCY
                        string rawId = GetStringSafe(row, "ID");
                        if (!string.IsNullOrWhiteSpace(rawId))
                        {
                            item.Id = rawId; // Engineering Toolbox format: Use the provided ID
                        }
                        else
                        {
                            // BOM Reader format: Generate the ID using the Level column
                            int currentLevel = GetIntSafe(row, "Level", "level") ?? 1;

                            // Increment the counter for the current level
                            if (levelCounters.ContainsKey(currentLevel))
                                levelCounters[currentLevel]++;
                            else
                                levelCounters[currentLevel] = 1;

                            // Reset counters for any deeper levels so they start back at 1
                            var keysToRemove = levelCounters.Keys.Where(k => k > currentLevel).ToList();
                            foreach (var k in keysToRemove) levelCounters.Remove(k);

                            // Construct the ID string (e.g., 001.002.001)
                            var idParts = new List<string>();
                            for (int i = 1; i <= currentLevel; i++)
                            {
                                if (levelCounters.TryGetValue(i, out int count))
                                    idParts.Add(count.ToString("D3")); // Pads with zeroes
                                else
                                    idParts.Add("001");
                            }
                            item.Id = string.Join(".", idParts);
                        }

                        if (!string.IsNullOrWhiteSpace(item.PartNo))
                            result.BomItems.Add(item);
                    }
                }
            }
            catch (IOException ex)
            {
                throw new Exception($"The file '{Path.GetFileName(filePath)}' is currently open in another program (like Excel). Please close it and try again.", ex);
            }

            return result;
        }

        public List<SchematicItem> ReadSchematic(string filePath)
        {
            var list = new List<SchematicItem>();
            using (var workbook = new XLWorkbook(filePath))
            {
                var ws = workbook.Worksheet(1);
                int headerRowIdx = -1;

                // Dynamically find where the actual data starts by scanning for "Subsystem"
                for (int i = 1; i <= 20; i++)
                {
                    if (ws.Cell(i, 1).GetString().Trim().Equals("Subsystem", System.StringComparison.OrdinalIgnoreCase))
                    {
                        headerRowIdx = i;
                        break;
                    }
                }

                if (headerRowIdx == -1) throw new System.Exception("Could not find 'Subsystem' header in schematic file.");

                var headerMap = new Dictionary<string, int>(System.StringComparer.OrdinalIgnoreCase);
                foreach (var cell in ws.Row(headerRowIdx).CellsUsed())
                {
                    string headerName = cell.GetString().Trim();
                    if (!string.IsNullOrWhiteSpace(headerName))
                        headerMap[headerName] = cell.Address.ColumnNumber;
                }

                int GetCol(params string[] aliases)
                {
                    foreach (var name in aliases)
                        if (headerMap.TryGetValue(name, out int colIndex)) return colIndex;
                    return -1;
                }

                int subCol = GetCol("Subsystem");
                int partCol = GetCol("Cable/Device K#", "Part Number");
                int descCol = GetCol("Description", "Desc");
                int qtyCol = GetCol("QTY", "Quantity");

                var rows = ws.RowsUsed().Skip(headerRowIdx);
                foreach (var row in rows)
                {
                    string part = partCol != -1 ? row.Cell(partCol).GetString().Trim() : "";
                    string sub = subCol != -1 ? row.Cell(subCol).GetString().Trim() : "";

                    // Skip completely empty rows
                    if (string.IsNullOrWhiteSpace(part) && string.IsNullOrWhiteSpace(sub)) continue;

                    list.Add(new SchematicItem
                    {
                        Subsystem = sub,
                        PartNumber = part,
                        Description = descCol != -1 ? row.Cell(descCol).GetString().Trim() : "",
                        Qty = qtyCol != -1 && row.Cell(qtyCol).TryGetValue<double>(out var q) ? q : 0
                    });
                }
            }
            return list;
        }

        private void Filter_TextChanged(object sender, System.Windows.Controls.TextChangedEventArgs e)
        {
            if (_reportData == null) return;

            // Escaping single quotes to prevent SQL syntax crashes
            string partFilter = PartNoFilter.Text.Trim().Replace("'", "''");
            string catFilter = CategoryFilter.Text.Trim().Replace("'", "''");

            var filters = new List<string>();

            // [Part Number] requires brackets because the column name contains a space
            if (!string.IsNullOrEmpty(partFilter))
                filters.Add($"[Part Number] LIKE '%{partFilter}%'");

            if (!string.IsNullOrEmpty(catFilter))
                filters.Add($"Category LIKE '%{catFilter}%'");

            // Combine filters and apply to the DataView
            _reportData.DefaultView.RowFilter = string.Join(" AND ", filters);
        }

        private bool FilterDifferences(object obj)
        {
            if (obj is not BomDifference diff) return false;

            bool matchesPart = string.IsNullOrWhiteSpace(SearchPartNo) || diff.PartNo.Contains(SearchPartNo, StringComparison.OrdinalIgnoreCase);
            bool matchesIssue = SearchIssueType == "All" || string.IsNullOrWhiteSpace(SearchIssueType) || diff.IssueType.Contains(SearchIssueType, StringComparison.OrdinalIgnoreCase);
            bool matchesAll = string.IsNullOrWhiteSpace(SearchAllColumns) ||
                              diff.PartNo.Contains(SearchAllColumns, StringComparison.OrdinalIgnoreCase) ||
                              diff.Description.Contains(SearchAllColumns, StringComparison.OrdinalIgnoreCase) ||
                              diff.IssueType.Contains(SearchAllColumns, StringComparison.OrdinalIgnoreCase) ||
                              diff.Details.Contains(SearchAllColumns, StringComparison.OrdinalIgnoreCase);

            return matchesPart && matchesIssue && matchesAll;
        }

        [RelayCommand]
        private async Task RunComparisonAsync()
        {
            if (FileQueue.Count < 2)
            {
                System.Windows.MessageBox.Show("Please add at least 2 files to the queue to compare.", "Insufficient Files", System.Windows.MessageBoxButton.OK, System.Windows.MessageBoxImage.Information);
                return;
            }

            IsBusy = true;
            StatusMessage = "Loading and comparing BOMs...";

            try
            {
                List<string> missingWarnings = new List<string>();

                var diffs = await Task.Run(() =>
                {
                    LoadedBoms.Clear();

                    // Loop through the dynamic queue and load everything
                    foreach (var fileItem in FileQueue)
                    {
                        var result = _excelService.ReadBom(fileItem.FilePath);

                        // We now extract the DisplayName (which is the Nickname if provided, or OriginalName if not)
                        string fileName = fileItem.DisplayName;

                        // The contingency here prevents a crash if a user accidentally nicknames two files the exact same thing
                        if (!LoadedBoms.ContainsKey(fileName))
                            LoadedBoms.Add(fileName, result.BomItems);
                        else
                            LoadedBoms.Add($"{fileName} ({Guid.NewGuid().ToString().Substring(0, 4)})", result.BomItems);

                        var missing = result.MissingColumns.Except(missingWarnings).ToList();
                        if (missing.Any()) missingWarnings.AddRange(missing);
                    }

                    // Pass the dictionary to our new N-way comparator
                    return _comparisonService.CompareBoms(LoadedBoms);
                });

                Differences.Clear();
                foreach (var d in diffs) Differences.Add(d);

                if (missingWarnings.Any())
                {
                    System.Windows.MessageBox.Show($"Warning: The following expected columns were not found and will be skipped:\n\n{string.Join(", ", missingWarnings)}",
                        "Missing Columns Detected", System.Windows.MessageBoxButton.OK, System.Windows.MessageBoxImage.Warning);
                }

                StatusMessage = $"Comparison complete. Found {diffs.Count} differences across {FileQueue.Count} files.";
            }
            catch (Exception ex)
            {
                System.Windows.MessageBox.Show(ex.Message, "Error Loading File", System.Windows.MessageBoxButton.OK, System.Windows.MessageBoxImage.Error);
                StatusMessage = "Error encountered during comparison.";
            }
            finally
            {
                IsBusy = false;
            }
        }
